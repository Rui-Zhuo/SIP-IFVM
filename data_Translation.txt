subroutine OutputHDF5Parallel(time,index_step)
use mpi
use hdf5
use common_u
use common_grid
use common_rank
use params
implicit none
real(8),intent(in) :: time
integer,intent(in) :: index_step
integer :: ierr
integer :: i,j,k
integer:: theta_indStart,theta_indEnd,ph_indStart,ph_indEnd,r_indStart,r_indEnd
integer :: Inttime_temp,hdf5index_r,hdf5index_theta,hdf5index_ph,ii,jj,kk
real(8) :: rho
real(8) :: vx,vy,vz
real(8) :: P
real(8) :: Bx,By,Bz
real(8) :: OUT_cadence 
real(8) :: time_temp
character(len=256) :: filnam
character(len=1)  :: cnyy
character(len=2) :: S2_Char
character(len=:),allocatable :: ctime
character(len=:),allocatable :: S1_Char
!=========================================================
! HDF5 variables
!=========================================================
integer(HID_T) :: file_id
integer(HID_T) :: dset_id
integer(HID_T) :: filespace
integer(HID_T) :: memspace
integer(HID_T) :: plist_id
integer(HID_T) :: dxpl_id
integer(HID_T) :: attr_id
integer(HID_T) :: attr_space
integer(HSIZE_T),dimension(4) :: dims
integer(HSIZE_T),dimension(4) :: count
integer(HSIZE_T),dimension(4) :: offset
integer(HSIZE_T),dimension(1) :: adims
!=========================================================
! local buffer
!=========================================================
real(8),allocatable :: Variables(:,:,:,:)
integer :: local_r
integer :: local_th
integer :: local_ph
integer :: r_indexglob
integer :: th_indexglob
integer :: ph_indexglob
!=========================================================
! filename
!=========================================================
  allocate(character(len=10)::ctime)
  allocate(character(len=10)::S1_Char)
  time_temp=time*100.d0
  Inttime_temp=int(time_temp)
  call INT2STR(Inttime_temp,S1_Char)
  if(len_trim(S1_Char)>=3) then
      S2_Char=S1_Char(len_trim(S1_Char)-1:len_trim(S1_Char))
      S1_Char=S1_Char(1:len_trim(S1_Char)-2)
  else if(len_trim(S1_Char)==2) then
      S2_Char=S1_Char(len_trim(S1_Char)-1:len_trim(S1_Char))
      S1_Char='0'
  else
      S2_Char='0'//S1_Char(len_trim(S1_Char):len_trim(S1_Char))
      S1_Char='0'
  endif
  !OUT_cadence=1.d-2
  !index_step=2
  !if((time-int(time))<OUT_cadence) then
  !    index_step=1
  !endif
  !index_step=2
  if(time_int==0) then
      ctime=S1_Char//'_'//S2_Char
  else
      ctime=S1_Char
  endif
  write(cnyy,"(i1.1)") nyy
  if(index_step==3)then
      filnam='DATA'//trim(folder_Suffix)//trim(ctime)//'_'//cnyy//'skip2.h5'
  else if(index_step==2)then
      filnam='DATA'//trim(folder_Suffix)//trim(ctime)//'_'//cnyy//'skip1.h5'
  else if(index_step==1)then
      filnam='DATA'//trim(folder_Suffix)//trim(ctime)//'_'//cnyy//'ori.h5'
  end if   
!write(cnyy,'(I0)') nyy
!write(filnam,'("DATA/Component_",A,".h5")') trim(cnyy)
!=========================================================
! local size
!=========================================================
  theta_indStart=1    
  theta_indEnd=N_th  
  ph_indStart=1  
  ph_indEnd=N_ph
  r_indStart=1  
  r_indEnd=N_r
  hdf5index_r=int((r_indEnd-r_indStart)/index_step)+1
  hdf5index_theta=int((theta_indEnd-theta_indStart)/index_step)+1
  hdf5index_ph=int((ph_indEnd-ph_indStart)/index_step)+1 
local_r  = hdf5index_r
local_th = hdf5index_theta
local_ph = hdf5index_ph
allocate(Variables(8,local_r,local_th,local_ph))
!=========================================================
! fill local physical cells
! no ghost cells
!=========================================================
do kk=1,local_ph
    k=(kk-1)*index_step+1
    do jj=1,local_th
        j=(jj-1)*index_step+1
        do ii=1,local_r
            i=(ii-1)*index_step+1
            rho = U(1,i,j,k)
            vx = U(2,i,j,k)/rho
            vy = U(3,i,j,k)/rho
            vz = U(4,i,j,k)/rho
            P = U(6,i,j,k)
            Bx = U(7,i,j,k)+B0(1,i,j,k)
            By = U(8,i,j,k)+B0(2,i,j,k)
            Bz = U(9,i,j,k)+B0(3,i,j,k)
            Variables(1,ii,jj,kk)=rho
            Variables(2,ii,jj,kk)=vx
            Variables(3,ii,jj,kk)=vy
            Variables(4,ii,jj,kk)=vz
            Variables(5,ii,jj,kk)=P
            Variables(6,ii,jj,kk)=Bx
            Variables(7,ii,jj,kk)=By
            Variables(8,ii,jj,kk)=Bz
        enddo
    enddo
enddo
!=========================================================
! global position
! calculated in grid_system()
!=========================================================
r_indexglob  = mpi_yinyang_comp_coord(l_sub_r)*local_r
th_indexglob = mpi_yinyang_comp_coord(l_sub_th)*local_th
ph_indexglob = mpi_yinyang_comp_coord(l_sub_ph)*local_ph
!=========================================================
! HDF5 initialize
!=========================================================
call h5open_f(ierr)
!=========================================================
! MPI-IO file access property
!=========================================================
call h5pcreate_f(H5P_FILE_ACCESS_F,plist_id,ierr)
call h5pset_fapl_mpio_f( &
        plist_id, &
        comm_yinyang_comp_carr, &
        MPI_INFO_NULL, &
        ierr)
!=========================================================
! create file collectively
!=========================================================
call h5fcreate_f( &
        trim(filnam), &
        H5F_ACC_TRUNC_F, &
        file_id, &
        ierr, &
        access_prp=plist_id)
call h5pclose_f(plist_id,ierr)
!=========================================================
! global dataset size
!
! Variables(8,Nr,Nth,Nph)
!
!=========================================================
dims(1)=8
dims(2)=pe_r_total*local_r
dims(3)=pe_th_total(nyy)*local_th
dims(4)=pe_ph_total(nyy)*local_ph
call h5screate_simple_f(4,dims,filespace,ierr)
!=========================================================
! create dataset
!=========================================================
call h5dcreate_f( &
        file_id, &
        "Variables", &
        H5T_NATIVE_DOUBLE, &
        filespace, &
        dset_id, &
        ierr)
call h5sclose_f(filespace,ierr)
!=========================================================
! select hyperslab
!=========================================================
call h5dget_space_f(dset_id,filespace,ierr)
offset(1)=0
offset(2)=r_indexglob
offset(3)=th_indexglob
offset(4)=ph_indexglob
count(1)=8
count(2)=local_r
count(3)=local_th
count(4)=local_ph
call h5sselect_hyperslab_f( &
        filespace, &
        H5S_SELECT_SET_F, &
        offset, &
        count, &
        ierr)
!=========================================================
! memory space
!=========================================================
call h5screate_simple_f(4,count,memspace,ierr)
!=========================================================
! collective write property
!=========================================================
call h5pcreate_f( &
        H5P_DATASET_XFER_F, &
        dxpl_id, &
        ierr)
call h5pset_dxpl_mpio_f( &
        dxpl_id, &
        H5FD_MPIO_COLLECTIVE_F, &
        ierr)
!=========================================================
! collective MPI write
!=========================================================
call h5dwrite_f( &
        dset_id, &
        H5T_NATIVE_DOUBLE, &
        Variables, &
        count, &
        ierr, &
        file_space_id = filespace, &
        mem_space_id  = memspace, &
        xfer_prp      = dxpl_id)
call h5pclose_f(dxpl_id,ierr)
!=========================================================
! close HDF5 objects
!=========================================================
call h5sclose_f(memspace,ierr)
call h5sclose_f(filespace,ierr)
call h5dclose_f(dset_id,ierr)
call h5fclose_f(file_id,ierr)
call h5close_f(ierr)
!=========================================================
!---------------------------------------
! wait until everyone has closed file
!---------------------------------------
call MPI_BARRIER(comm_yinyang_comp_carr,ierr)
!---------------------------------------
! reopen file serially
!---------------------------------------
if( mpi_yinyang_comp_coord(l_sub_r)==0 .and. &
    mpi_yinyang_comp_coord(l_sub_th)==0 .and. &
    mpi_yinyang_comp_coord(l_sub_ph)==0 ) then
    call h5open_f(ierr)
    call h5fopen_f(trim(filnam),H5F_ACC_RDWR_F,file_id,ierr)
    call h5dopen_f(file_id,"Variables",dset_id,ierr)
    adims(1)=1
    call h5screate_simple_f(1,adims,attr_space,ierr)
    call h5acreate_f(dset_id,"simulation_time", &
                     H5T_NATIVE_DOUBLE,attr_space,attr_id,ierr)
    call h5awrite_f(attr_id,H5T_NATIVE_DOUBLE,time,adims,ierr)
    call h5aclose_f(attr_id,ierr)
    call h5acreate_f(dset_id,"nyy_component", &
                     H5T_NATIVE_INTEGER,attr_space,attr_id,ierr)
    call h5awrite_f(attr_id,H5T_NATIVE_INTEGER,nyy,adims,ierr)
    call h5aclose_f(attr_id,ierr)
    call h5sclose_f(attr_space,ierr)
    call h5dclose_f(dset_id,ierr)
    call h5fclose_f(file_id,ierr)
    call h5close_f(ierr)
endif 
! release memory
!=========================================================
deallocate(Variables)
end subroutine OutputHDF5Parallel
